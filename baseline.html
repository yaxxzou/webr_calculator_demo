webRSession = {
  let session = null
  const packages = new Map()
  const startR = async () => {
    const {WebR} = await import(`https://webr.r-wasm.org/v${webrVersion}/webr.mjs`)
    const webR = new WebR()
    await webR.init()
    return {webR, rVersion: await webR.evalRString("R.version.string")}
  }
  // if starting R or installing a package fails, the next click tries again
  const getR = () => session ??= startR().catch(e => { session = null; throw e })
  const install = async pkg => {
    const {webR, rVersion} = await getR()
    await webR.installPackages([pkg], {quiet: true})
    await webR.evalRVoid(`suppressPackageStartupMessages(library(${pkg}))`)
    return {webR, rVersion, version: await webR.evalRString(`as.character(packageVersion("${pkg}"))`)}
  }
  return {
    // whether a package has been requested, so that the first, slower run can say what it is doing
    started: pkg => packages.has(pkg),
    load: pkg => {
      if (!packages.has(pkg)) packages.set(pkg, install(pkg).catch(e => { packages.delete(pkg); throw e }))
      return packages.get(pkg)
    }
  }
}

statcheckR = ({
  started: () => webRSession.started("statcheck"),
  get: () => webRSession.load("statcheck").then(({webR, rVersion, version}) =>
    ({webR, rVersion, statcheckVersion: version, version: `statcheck ${version}, ${rVersion}, running in webR ${webrVersion}`}))
})

async function runStatcheck(lines, oneTailed) {
  const {webR, ...versions} = await statcheckR.get()
  const shelter = await new webR.Shelter()
  try {
    // the lines are passed to R as a character vector, never pasted into the code, and named by their line number
    // so that each result statcheck returns can be matched to the line it came from
    const res = await shelter.evalR(`
      names(lines) <- seq_along(lines)
      res <- statcheck(lines, OneTailedTests = one_tailed, messages = FALSE)
      if (is.null(res)) data.frame() else as.data.frame(res)
    `, {env: {lines, one_tailed: oneTailed}})
    const rows = await res.toD3()
    return {lines, rows, oneTailed, runAt: new Date(), ...versions}
  } finally {
    shelter.purge()
  }
}

function statcheckTable(run) {
  const {lines, rows, version, oneTailed} = run
  // one row per result found, and one per line in which statcheck found no result
  const verdict = r => r.error === null ? ["Could not check", "na"]
    : r.decision_error ? ["Grossly inconsistent", "bad"]
    : r.error ? ["Inconsistent", "warn"]
    : ["Consistent", "ok"]
  const comparison = r => r.p_comp === "ns" ? "ns" : `p ${r.p_comp} ${formatReported(r)}`
  const body = lines.flatMap((line, i) => {
    const found = rows.filter(r => r.source === String(i + 1))
    if (found.length === 0) return [html`<tr class="sc-row-none"><td>${i + 1}</td><td>${line}</td><td></td><td></td><td>Not recognised</td></tr>`]
    return found.map(r => {
      const [label, kind] = verdict(r)
      return html`<tr><td>${i + 1}</td><td>${r.raw}</td><td>${comparison(r)}</td><td>${r.computed_p === null ? "" : formatP(r.computed_p)}</td><td><span class="sc-verdict sc-verdict-${kind}">${label}</span></td></tr>`
    })
  })
  const counts = d3.rollup(rows, v => v.length, r => verdict(r)[0])
  const summary = ["Consistent", "Inconsistent", "Grossly inconsistent"].map(k => `${counts.get(k) ?? 0} ${k.toLowerCase()}`).join(", ")
  const unread = lines.filter((l, i) => !rows.some(r => r.source === String(i + 1))).length
  return html`<div>
    <div class="table-responsive"><table class="table table-sm sc-table">
      <thead><tr><th>Line</th><th>Result as read by statcheck</th><th>Reported</th><th>Recalculated <em>p</em></th><th>Verdict</th></tr></thead>
      <tbody>${body}</tbody>
    </table></div>
    <div class="sc-status">${rows.length} result${rows.length === 1 ? "" : "s"} found: ${summary}.${unread > 0 ? ` ${unread} line${unread === 1 ? " was" : "s were"} not recognised.` : ""} Tests treated as ${oneTailed ? "one-sided" : "two-sided"}, with α = .05. ${version}.</div>
    <div class="sb-buttons sc-download">${downloadButton(run, verdict)}</div>
  </div>`
}

function downloadButton(run, verdict) {
  // a button that saves the inputs, statcheck's output, and the versions used as a CSV file. one row per result found,
  // and one per line in which statcheck found no result, in the same order as the table
  const {lines, rows, oneTailed, runAt, statcheckVersion, rVersion} = run
  const columns = ["line", "input", "raw", "test_type", "df1", "df2", "test_comp", "test_value", "p_comp", "reported_p", "computed_p", "error", "decision_error", "verdict",
    "one_tailed_tests", "alpha", "statcheck_version", "r_version", "webr_version", "run_at"]
  const settings = {one_tailed_tests: oneTailed, alpha: 0.05, statcheck_version: statcheckVersion, r_version: rVersion, webr_version: webrVersion, run_at: runAt.toISOString()}
  const records = lines.flatMap((line, i) => {
    const found = rows.filter(r => r.source === String(i + 1))
    if (found.length === 0) return [{line: i + 1, input: line, verdict: "Not recognised", ...settings}]
    // the reported p-value is kept as it was written, so that its decimals are not lost
    return found.map(r => ({...r, line: i + 1, input: line, reported_p: r.p_comp === "ns" ? "" : formatReported(r), verdict: verdict(r)[0], ...settings}))
  })
  return csvDownloadButton(columns, records, "statcheck_results", runAt)
}

function csvDownloadButton(columns, records, prefix, runAt) {
  // a button that saves records (objects with the given columns) as a CSV file, named after the prefix and the local
  // date and time of the run, e.g. statcheck_results_2026-10-16_14-05-09.csv
  const cell = x => {
    if (x === null || x === undefined) return ""
    const text = typeof x === "boolean" ? (x ? "TRUE" : "FALSE") : String(x)
    return /[",\n\r]/.test(text) ? `"${text.replace(/"/g, '""')}"` : text
  }
  const csv = [columns.join(","), ...records.map(r => columns.map(c => cell(r[c])).join(","))].join("\r\n")
  const pad = n => String(n).padStart(2, "0")
  const d = runAt
  const name = `${prefix}_${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}_${pad(d.getHours())}-${pad(d.getMinutes())}-${pad(d.getSeconds())}.csv`
  const button = html`<button class="btn btn-quarto">Download results as CSV</button>`
  button.onclick = () => {
    // the byte order mark lets spreadsheet programs read symbols such as χ and − correctly
    const url = URL.createObjectURL(new Blob(["\ufeff" + csv], {type: "text/csv;charset=utf-8"}))
    const link = html`<a href="${url}" download="${name}"></a>`
    document.body.append(link)
    link.click()
    link.remove()
    setTimeout(() => URL.revokeObjectURL(url), 1000)
  }
  return button
}

function formatReported(r) {
  // the reported p-value as it was written, so that its decimals are kept (statcheck returns it as a number, in which
  // .000 becomes 0 and .040 becomes 0.04). falls back to the number, without its leading zero
  const written = r.raw.match(/p\s*[=<>]\s*(0?\.\d+|\d+(?:\.\d+)?)\s*$/i)
  return written ? written[1] : String(r.reported_p).replace(/^0\./, ".")
}

function parseResult(text) {
  // reads a result reported in APA style, e.g. "t(28) = 2.0, p < .05" or "χ2(1, N = 226) = 6.90, p < .01", and keeps
  // the number of decimals of the test statistic and the p-value, which statcheck needs to account for rounding.
  // returns {message} if the text cannot be read
  const s = String(text).replace(/[−–—]/g, "-").replace(/\s+/g, " ").trim()
  const m = s.match(/^(t|f|r|z|χ2|χ²|chi2|chi²|x2)\s*(?:\(([^)]*)\))?\s*=\s*(-?\d*\.?\d+)\s*[,;]\s*p\s*([=<>])\s*(\d*\.?\d+)$/i)
  if (!m) return {message: "Type a result in APA style, such as t(28) = 2.0, p < .05. The test can be t, F, r, χ2, or z, and the p-value can be reported with =, <, or >."}
  const type = {t: "t", f: "F", r: "r", z: "z"}[m[1].toLowerCase()] ?? "chi2"
  // degrees of freedom are the plain numbers in the brackets, so that the N in "χ2(1, N = 226)" is left out
  const dfs = (m[2] ?? "").split(",").map(x => x.trim()).filter(x => /^\d*\.?\d+$/.test(x)).map(Number)
  const decimals = x => (x.split(".")[1] ?? "").length
  const stat = Number(m[3]), pRep = Number(m[5])
  const result = {type, stat, statText: m[3], statDec: decimals(m[3]), pComp: m[4], pRep, pText: m[5], pDec: decimals(m[5]), df1: NaN, df2: NaN}
  if (type === "t" || type === "r") result.df2 = dfs[0]
  if (type === "F") { result.df1 = dfs[0]; result.df2 = dfs[1] }
  if (type === "chi2") result.df1 = dfs[0]
  const needs = {t: ["df2"], r: ["df2"], F: ["df1", "df2"], chi2: ["df1"], z: []}[type]
  if (needs.some(k => !(result[k] > 0))) return {message: type === "F" ? "An F test needs two degrees of freedom, e.g. F(2, 40) = 3.50, p = .040." : `This test needs its degrees of freedom in brackets, e.g. ${type === "chi2" ? "χ2(1)" : type + "(28)"}.`}
  if (type === "r" && Math.abs(stat) >= 1) return {message: "A correlation has to be between −1 and 1."}
  if ((type === "F" || type === "chi2") && stat < 0) return {message: "An F or χ2 statistic cannot be negative."}
  if (pRep > 1) return {message: "A p-value cannot be larger than 1."}
  return result
}

function pFrom(type, stat, df1, df2, twoTailed) {
  // the p-value of a test statistic, by the same distributions as statcheck. t and F through the regularised
  // incomplete beta function, chi-squared and z through the regularised upper incomplete gamma function. a z
  // statistic squared is chi-squared with one degree of freedom. a correlation is converted to t first
  let p
  if (type === "t" || type === "r") {
    const t = type === "r" ? stat * Math.sqrt(df2 / (1 - stat * stat)) : stat
    p = betaI(df2 / (df2 + t * t), df2 / 2, 0.5)
  }
  else if (type === "z") p = gammaQ(0.5, stat * stat / 2)
  else if (type === "F") p = stat <= 0 ? 1 : betaI(df2 / (df2 + df1 * stat), df2 / 2, df1 / 2)
  else p = stat <= 0 ? 1 : gammaQ(df1 / 2, stat / 2)
  return (type === "t" || type === "r" || type === "z") && !twoTailed ? p / 2 : p
}

function invertP(pOf, target, upper) {
  // the test statistic whose p-value is target, by bisection: the p-value only decreases as the statistic increases
  let lower = 0
  if (upper === null) { upper = 1; while (pOf(upper) > target && upper < 1e6) upper *= 2 }
  if (pOf(upper) > target) return null
  for (let i = 0; i < 100; i++) {
    const middle = (lower + upper) / 2
    if (pOf(middle) > target) lower = middle; else upper = middle
  }
  return (lower + upper) / 2
}

function statSymbol(s) {

}

function formatP(p) {
  // four decimals, without the leading zero, as in APA style; very small values in scientific notation
  if (p === 0) return "0"
  return p < 0.0001 ? p.toExponential(1) : p.toFixed(4).replace(/^0/, "")
}

function logTicks(yMin) {

}

function betaI(x, a, b) {
  // regularised incomplete beta function, by continued fraction
  if (x <= 0) return 0
  if (x >= 1) return 1
  const front = Math.exp(logGamma(a + b) - logGamma(a) - logGamma(b) + a * Math.log(x) + b * Math.log(1 - x))
  const cf = (x, a, b) => {
    let c = 1, d = 1 - (a + b) * x / (a + 1)
    if (Math.abs(d) < 1e-300) d = 1e-300
    d = 1 / d
    let h = d
    for (let m = 1; m < 500; m++) {
      let aa = m * (b - m) * x / ((a + 2 * m - 1) * (a + 2 * m))
      d = 1 + aa * d; if (Math.abs(d) < 1e-300) d = 1e-300
      c = 1 + aa / c; if (Math.abs(c) < 1e-300) c = 1e-300
      d = 1 / d
      h *= d * c
      aa = -(a + m) * (a + b + m) * x / ((a + 2 * m) * (a + 2 * m + 1))
      d = 1 + aa * d; if (Math.abs(d) < 1e-300) d = 1e-300
      c = 1 + aa / c; if (Math.abs(c) < 1e-300) c = 1e-300
      d = 1 / d
      const del = d * c
      h *= del
      if (Math.abs(del - 1) < 1e-14) break
    }
    return h
  }
  return x < (a + 1) / (a + b + 2) ? front * cf(x, a, b) / a : 1 - front * cf(1 - x, b, a) / b
}

function gammaQ(a, x) {
  // regularised upper incomplete gamma function: series for small x, continued fraction otherwise. the upper tail is
  // calculated directly in the continued fraction, so that very small p-values keep their precision
  if (x <= 0) return 1
  const front = Math.exp(-x + a * Math.log(x) - logGamma(a))
  if (x < a + 1) {
    let term = 1 / a, sum = term
    for (let n = 1; n < 500; n++) { term *= x / (a + n); sum += term; if (Math.abs(term) < Math.abs(sum) * 1e-14) break }
    return 1 - sum * front
  }
  let b = x + 1 - a, c = 1e300, d = 1 / b, h = d
  for (let i = 1; i < 500; i++) {
    const an = -i * (i - a)
    b += 2
    d = an * d + b; if (Math.abs(d) < 1e-300) d = 1e-300
    c = b + an / c; if (Math.abs(c) < 1e-300) c = 1e-300
    d = 1 / d
    const del = d * c
    h *= del
    if (Math.abs(del - 1) < 1e-14) break
  }
  return front * h
}

function logGamma(x) {
  // Lanczos approximation
  const c = [76.18009172947146, -86.50532032941677, 24.01409824083091, -1.231739572450155, 0.001208650973866179, -0.000005395239384953]
  let y = x, sum = 1.000000000190015
  for (const cj of c) sum += cj / ++y
  return -(x + 5.5) + (x + 0.5) * Math.log(x + 5.5) + Math.log(2.5066282746310005 * sum / x)
}

async function runEscalc(rows, correct) {
  const {webR, rVersion, version} = await webRSession.load("metafor")
  const valid = rows.filter(r => r.valid)
  const shelter = await new webR.Shelter()
  try {
    // the values are passed to R as numeric vectors, never pasted into the code
    const col = k => valid.map(r => r[k])
    const res = await shelter.evalR(`
      es <- summary(escalc(measure = "SMD", m1i = m1, sd1i = sd1, n1i = n1, m2i = m2, sd2i = sd2, n2i = n2, correct = correct))
      data.frame(yi = as.numeric(es$yi), vi = as.numeric(es$vi), sei = es$sei, ci.lb = es$ci.lb, ci.ub = es$ci.ub)
    `, {env: {m1: col("m1"), sd1: col("sd1"), n1: col("n1"), m2: col("m2"), sd2: col("sd2"), n2: col("n2"), correct}})
    const out = await res.toD3()
    valid.forEach((r, i) => Object.assign(r, out[i]))
    return {rows, correct, runAt: new Date(), metaforVersion: version, rVersion}
  } finally {
    shelter.purge()
  }
}

function ttChecked(raw) {
  // reads and checks the values in the calculator. the means and SDs are required. the number of decimal places of each
  // kind of value is taken from the values as typed, and if they differ the fewest is used, which is the most lenient
  const problems = []
  const clean = s => s.replace(/−/g, "-")
  const number = s => /^[-+]?(\d+\.?\d*|\.\d+)$/.test(clean(s)) ? Number(clean(s)) : NaN
  const decimals = s => s.match(/\.(\d*)$/)?.[1].length ?? 0
  const groups = raw.groups.map((g, i) => {
    const out = {m: number(g.m), sd: number(g.sd), n: number(g.n)}
    if (!Number.isFinite(out.m)) problems.push(`Group ${i + 1}: enter its mean.`)
    if (!Number.isFinite(out.sd) || out.sd <= 0) problems.push(`Group ${i + 1}: its SD must be a number above zero.`)
    if (!Number.isInteger(out.n) || out.n < 2) problems.push(`Group ${i + 1}: its n must be a whole number of at least 2.`)
    return out
  })
  const optional = (s, label) => {
    if (s === "") return null
    const x = number(s)
    if (!Number.isFinite(x)) problems.push(`The reported ${label} is not a number.`)
    return x
  }
  const p = optional(raw.p, "p-value"), d = optional(raw.d, "SMD"), lower = optional(raw.lower, "lower bound of the CI"), upper = optional(raw.upper, "upper bound of the CI")
  if (p !== null && (p < 0 || p > 1)) problems.push("The reported p-value must be between 0 and 1.")
  if ((lower === null) !== (upper === null)) problems.push("Enter both bounds of the CI, or neither.")
  else if (lower !== null && lower >= upper) problems.push("The lower bound of the CI must be smaller than its upper bound.")
  const fewest = values => d3.min(values.filter(v => v !== ""), decimals)
  return {
    problems, groups, p, d, lower, upper, raw,
    mDigits: fewest(raw.groups.map(g => g.m)), sdDigits: fewest(raw.groups.map(g => g.sd)),
    pDigits: p === null ? null : decimals(raw.p),
    dDigits: d === null && lower === null ? null : fewest([raw.d, raw.lower, raw.upper]),
    pOp: raw.pOp, alternative: raw.alternative === "one.sided" ? oneSided(groups, raw.direction) : ["two.sided"],
    direction: raw.direction, seConfusion: raw.seConfusion,
    slow: raw.direction === "both" && raw.seConfusion
  }
}

function oneSided(groups, direction) {
  // a reported one-sided p-value is almost always in the direction of the observed difference. including the other
  // direction too would allow p-values close to 1, and make almost any reported p-value consistent. when either
  // difference is allowed, both tails are needed, because recalc applies each alternative to both differences
  if (direction === "both") return ["less", "greater"]
  const diff = direction === "m1_minus_m2" ? groups[0].m - groups[1].m : groups[1].m - groups[0].m
  return [diff >= 0 ? "greater" : "less"]
}

async function runTTest(input) {
  const {webR, rVersion, recalcVersion} = await recalcR.get()
  const shelter = await new webR.Shelter()
  try {
    // the values are passed to R as vectors, never pasted into the code. values that were not entered are passed as
    // NaN, which R turns into NULL
    const none = [NaN], opt = x => x === null ? none : [x]
    const [g1, g2] = input.groups
    const res = await shelter.evalR(`
      opt <- function(x) if (all(is.na(x))) NULL else x
      res <- recalc_env$recalc_independent_t(
        m1 = m1, m2 = m2, sd1 = sd1, sd2 = sd2, n1 = n1, n2 = n2,
        m_digits = m_digits, sd_digits = sd_digits, rounding = "either",
        p = opt(p), p_digits = opt(p_digits), p_operator = p_operator, alternative = alternative,
        d = opt(d), d_ci_lower = opt(lower), d_ci_upper = opt(upper), d_digits = opt(d_digits),
        direction = direction, include_se_sd_confusion = se_confusion)
      list(reproduced = as.data.frame(res$reproduced), p = res$p_method_intervals,
           d = res$d_method_intervals, ci = res$ci_method_intervals)
    `, {env: {
      m1: g1.m, m2: g2.m, sd1: g1.sd, sd2: g2.sd, n1: g1.n, n2: g2.n, m_digits: input.mDigits, sd_digits: input.sdDigits,
      p: opt(input.p), p_digits: opt(input.pDigits), p_operator: input.pOp, alternative: input.alternative,
      d: opt(input.d), lower: opt(input.lower), upper: opt(input.upper), d_digits: opt(input.dDigits),
      direction: input.direction, se_confusion: input.seConfusion
    }})
    const part = async name => (await res.get(name)).toD3()
    const [reproduced] = await part("reproduced")
    return {input, reproduced, pMethods: await part("p"), dMethods: await part("d"), ciMethods: await part("ci"),
      recalcVersion, rVersion, runAt: new Date()}
  } finally {
    shelter.purge()
  }
}

    p: [{student_t: "Student's t", welch_t: "Welch's t"}, {"two.sided": "two-sided", less: "one-sided (less)", greater: "one-sided (greater)"}, {m1_minus_m2: "1 − 2", m2_minus_m1: "2 − 1"}, {sd: "as SDs", se_converted: "SDs from SEs"}],

function orChecked(raw) {
  // reads and checks the values in the calculator. the counts must be whole numbers, and each group must have at least
  // one participant. the number of decimal places of the OR and its CI is taken from the values as typed, the fewest if
  // they differ
  const problems = []
  const clean = s => s.replace(/−/g, "-")
  const number = s => /^[-+]?(\d+\.?\d*|\.\d+)$/.test(clean(s)) ? Number(clean(s)) : NaN
  const decimals = s => s.match(/\.(\d*)$/)?.[1].length ?? 0
  const names = raw.names.map((n, i) => n || `Group ${i + 1}`)
  const cells = raw.cells.map((row, i) => row.map((c, j) => {
    const x = number(c)
    if (!Number.isInteger(x) || x < 0) problems.push(`${names[i]}: the number ${j === 0 ? "with" : "without"} the event must be a whole number of 0 or more.`)
    return x
  }))
  cells.forEach((row, i) => { if (row.every(Number.isInteger) && row[0] + row[1] === 0) problems.push(`${names[i]}: the group has no participants.`) })
  const optional = (s, label) => {
    if (s === "") return null
    const x = number(s)
    if (!Number.isFinite(x)) problems.push(`The reported ${label} is not a number.`)
    return x
  }
  const p = optional(raw.p, "p-value"), or = optional(raw.or, "odds ratio"), lower = optional(raw.lower, "lower bound of the CI"), upper = optional(raw.upper, "upper bound of the CI")
  if (p !== null && (p < 0 || p > 1)) problems.push("The reported p-value must be between 0 and 1.")
  if ([or, lower, upper].some(x => x !== null && x <= 0)) problems.push("An odds ratio and its CI must be above zero.")
  if (lower !== null && upper !== null && lower >= upper) problems.push("The lower bound of the CI must be smaller than its upper bound.")
  const fewest = values => d3.min(values.filter(v => v !== ""), decimals)
  return {
    problems, names, cells, p, or, lower, upper, raw, pOp: raw.pOp,
    pDigits: p === null ? null : decimals(raw.p),
    orDigits: or === null && lower === null && upper === null ? null : fewest([raw.or, raw.lower, raw.upper])
  }
}

async function runOddsRatio(input) {
  const {webR, rVersion, recalcVersion} = await recalcR.get()
  const shelter = await new webR.Shelter()
  try {
    // the values are passed to R as vectors, never pasted into the code. values that were not entered are passed as
    // NaN, which R turns into NULL. the counts are in row order: group 1 with and without the event, then group 2
    const none = [NaN], opt = x => x === null ? none : [x]
    const res = await shelter.evalR(`
      opt <- function(x) if (all(is.na(x))) NULL else x
      res <- recalc_env$recalc_oddsratio(
        counts = counts, n_rows = 2, n_cols = 2,
        odds_ratio = opt(or), or_ci_lower = opt(lower), or_ci_upper = opt(upper),
        or_digits = if (is.null(opt(or_digits))) 2L else as.integer(or_digits),
        p = opt(p), p_digits = opt(p_digits), p_operator = p_operator)
      list(reproduced = as.data.frame(res$reproduced), p = as.data.frame(res$method_intervals), or = as.data.frame(res$or_results))
    `, {env: {
      counts: input.cells.flat(), or: opt(input.or), lower: opt(input.lower), upper: opt(input.upper), or_digits: opt(input.orDigits),
      p: opt(input.p), p_digits: opt(input.pDigits), p_operator: input.pOp
    }})
    const part = async name => (await res.get(name)).toD3()
    const [reproduced] = await part("reproduced")
    return {input, reproduced, pMethods: await part("p"), orRows: await part("or"), recalcVersion, rVersion, runAt: new Date()}
  } finally {
    shelter.purge()
  }
}

    await webR.evalRVoid(`
      recalc_env <- new.env(parent = globalenv())
      for (src in sources) eval(parse(text = src), envir = recalc_env)
      assign("recalc_env", recalc_env, envir = globalenv())
    `, {env: {sources}})
    return {webR, rVersion, recalcVersion: description.match(/^Version:\s*(\S+)/m)?.[1] ?? "unknown"}
  }
  return {
    started: () => ready !== null,
    // if starting R or downloading recalc fails, the next click tries again
    get: () => ready ??= start().catch(e => { ready = null; throw e })
  }
}

function sgChecked(raw) {
  // reads and checks the values in the calculator. sample sizes must be whole numbers. the means and SDs are only used
  // if every subgroup has one, and SDs only together with means. the number of decimal places of the means and SDs is
  // taken from the values as typed, unless it was chosen, and if they differ the fewest is used, which is the most lenient
  const problems = [], notes = []
  const number = s => {
    const t = s.replace(/−/g, "-")
    return /^[-+]?(\d+\.?\d*|\.\d+)$/.test(t) ? Number(t) : NaN
  }
  const decimals = s => s.match(/\.(\d*)$/)?.[1].length ?? 0
  const rows = raw.rows.map((r, i) => ({...r, row: i + 1})).filter(r => r.label !== "" || r.n !== "" || r.m !== "" || r.sd !== "")
  const k = rows.length
  rows.forEach(r => {
    r.label = r.label || `Subgroup ${r.row}`
    r.nValue = number(r.n)
    r.mValue = r.m === "" ? null : number(r.m)
    r.sdValue = r.sd === "" ? null : number(r.sd)
    if (!Number.isInteger(r.nValue) || r.nValue < 1) problems.push(`Row ${r.row}: the n must be a whole number of at least 1.`)
    if (Number.isNaN(r.mValue)) problems.push(`Row ${r.row}: the mean is not a number.`)
    if (Number.isNaN(r.sdValue)) problems.push(`Row ${r.row}: the SD is not a number.`)
    else if (r.sdValue < 0) problems.push(`Row ${r.row}: the SD cannot be negative.`)
  })
  if (k === 0) problems.push("Enter at least one subgroup.")
  const withMean = rows.filter(r => r.m !== "").length, withSd = rows.filter(r => r.sd !== "").length
  // the rounding of every mean and SD is propagated by checking every combination of the ends of their rounding
  // intervals, which doubles with each value, and webR is much slower than R on a computer. so the number of subgroups
  // is limited to what takes no more than several seconds
  if (withSd > 0 && k > 6) problems.push("With SDs, the calculator handles up to 6 subgroups, because the time it takes doubles with every rounded value.")
  else if (k > 12) problems.push("The calculator handles up to 12 subgroups, because the time it takes doubles with every rounded value.")
  if (withMean > 0 && withMean < k) problems.push("Enter a mean for every subgroup, or leave the column empty.")
  if (withSd > 0 && withSd < k) problems.push("Enter an SD for every subgroup, or leave the column empty.")
  if (withSd > 0 && withMean === 0) problems.push("The SDs can only be used together with the means.")
  const useMeans = k > 0 && withMean === k, useSds = useMeans && withSd === k

  const o = raw.overall
  const overall = {
    n: o.n === "" ? null : number(o.n),
    m: o.m === "" ? null : number(o.m),
    sd: o.sd === "" ? null : number(o.sd),
    text: o
  }
  if (overall.n !== null && (!Number.isInteger(overall.n) || overall.n < 1)) problems.push("The whole sample's N must be a whole number of at least 1.")
  if (Number.isNaN(overall.m)) problems.push("The whole sample's mean is not a number.")
  if (Number.isNaN(overall.sd)) problems.push("The whole sample's SD is not a number.")
  else if (overall.sd < 0) problems.push("The whole sample's SD cannot be negative.")
  if (k === 1 && overall.n === null) problems.push("With only one subgroup, enter the whole sample's N, so that the calculator can work out what an unreported subgroup would need to be.")

  const scale = {min: raw.scaleMin === "" ? null : number(raw.scaleMin), max: raw.scaleMax === "" ? null : number(raw.scaleMax)}
  if (Number.isNaN(scale.min) || Number.isNaN(scale.max)) problems.push("The scale's minimum and maximum must be numbers.")
  else if ((scale.min === null) !== (scale.max === null)) problems.push("Enter both the scale's minimum and its maximum, or neither.")
  else if (scale.min !== null && scale.min >= scale.max) problems.push("The scale's minimum must be smaller than its maximum.")
  const useScale = scale.min !== null && scale.max !== null

  // values that were entered but cannot be used are pointed out, rather than silently ignored
  if (overall.m !== null && !useMeans) notes.push("The whole sample's mean was not checked, because the subgroups have no means.")
  if (overall.sd !== null && !useSds) notes.push("The whole sample's SD was not checked, because the subgroups have no SDs.")
  const typedDigits = values => values.length === 0 ? null : d3.min(values, decimals)
  const meanDigits = raw.meanDigits === "auto" ? typedDigits([...rows.map(r => r.m), ...(overall.m === null ? [] : [o.m])].filter(s => s !== "")) : Number(raw.meanDigits)
  const sdDigits = raw.sdDigits === "auto" ? typedDigits([...rows.map(r => r.sd), ...(overall.sd === null ? [] : [o.sd])].filter(s => s !== "")) : Number(raw.sdDigits)

  // what can be calculated: the whole sample from two or more subgroups, and, if the reported N is larger than the
  // sum of the subgroups' ns, the subgroup that would be missing
  const sumN = d3.sum(rows, r => r.nValue)
  const total = k >= 2
  const missing = overall.n !== null && sumN < overall.n
  const missingMean = missing && useMeans && overall.m !== null
  const missingSd = missingMean && useSds && overall.sd !== null
  if (missing && useMeans && overall.m === null) notes.push("Enter the whole sample's mean to work out the mean of the unreported subgroup.")
  if (missingMean && useSds && overall.sd === null) notes.push("Enter the whole sample's SD to work out the SD of the unreported subgroup.")
  if (useScale && !missingMean) notes.push("The scale's minimum and maximum are only used to check an unreported subgroup's mean and SD, so they were not used.")
  return {
    problems, notes, rows, overall, scale, useScale, useMeans, useSds, meanDigits, sdDigits, rounding: raw.rounding,
    sumN, total, missing, missingMean, missingSd, needsR: (total && useMeans) || missingMean,
    slow: (useSds && k >= 5) || k >= 10
  }
}

async function runSubgroupChecks(input) {
  const {webR, rVersion, recalcVersion} = await recalcR.get()
  const shelter = await new webR.Shelter()
  try {
    // the values are passed to R as vectors, never pasted into the code. webR cannot pass an empty array, so values that
    // were not entered or are not used are passed as NaN, which R turns into NULL (entered values are never NaN)
    const {rows, overall, scale} = input
    const none = [NaN]
    const opt = x => x === null ? none : [x]
    const res = await shelter.evalR(`
      opt <- function(x) if (all(is.na(x))) NULL else x
      total <- if (run_total) recalc_env$recalc_total_from_subgroups(
        subgroup_ns = ns, subgroup_means = ms, subgroup_sds = opt(sds),
        overall_n = opt(N), overall_mean = opt(M), overall_sd = opt(SD),
        n_digits = Inf, mean_digits = mean_digits, sd_digits = opt(sd_digits), rounding = rounding)
      missing <- if (run_missing) suppressWarnings(recalc_env$recalc_missing_subgroup(
        reported_ns = ns, reported_means = ms, reported_sds = opt(sds_missing),
        overall_n = N, overall_mean = M, overall_sd = opt(SD_missing),
        scale_min = opt(scale_min), scale_max = opt(scale_max),
        n_digits = Inf, mean_digits = mean_digits, sd_digits = opt(sd_digits), rounding = rounding))
      parts <- Filter(Negate(is.null), list(total = total, missing = missing))
      do.call(rbind, lapply(names(parts), function(p) cbind(part = p, as.data.frame(parts[[p]]))))
    `, {env: {
      run_total: input.total && input.useMeans, run_missing: input.missingMean,
      ns: rows.map(r => r.nValue), ms: rows.map(r => r.mValue),
      sds: input.useSds ? rows.map(r => r.sdValue) : none, sds_missing: input.missingSd ? rows.map(r => r.sdValue) : none,
      N: opt(overall.n), M: input.useMeans ? opt(overall.m) : none, SD: input.useSds ? opt(overall.sd) : none, SD_missing: input.missingSd ? [overall.sd] : none,
      scale_min: input.useScale ? [scale.min] : none, scale_max: input.useScale ? [scale.max] : none,
      mean_digits: input.meanDigits ?? 0, sd_digits: input.useSds ? [input.sdDigits] : none, rounding: input.rounding
    }})
    return sgCalculate(input, await res.toD3(), {recalcVersion, rVersion})
  } finally {
    shelter.purge()
  }
}

function sgCalculate(input, rRows, versions = null) {
  // gathers the results: the whole-sample checks and the implied missing subgroup, each as a list of rows for the
  // N (or n), the mean, and the SD. sample sizes are exact counts, so their checks need no R: R's results for them are
  // identical, and are only used when R was run anyway
  const {rows, overall} = input
  const fromR = (part, prefix) => rRows.find(r => r.part === part && r.check.startsWith(prefix))
  const value = x => (x === null || x === undefined) ? NaN : x
  const totals = [], missing = []
  if (input.total) {
    totals.push({quantity: "N", digits: null, reported: overall.text.n, reportedLower: value(overall.n), reportedUpper: value(overall.n),
      lower: input.sumN, upper: input.sumN, consistent: overall.n === null ? null : overall.n === input.sumN})
    for (const [quantity, prefix, digits] of [["Mean", "E1: M", input.meanDigits], ["SD", "E1: SD", input.sdDigits]]) {
      const r = fromR("total", prefix)
      if (r) totals.push({quantity, digits, reported: quantity === "Mean" ? overall.text.m : overall.text.sd,
        reportedLower: value(r.reported_lower), reportedUpper: value(r.reported_upper),
        lower: value(r.recalculated_lower), upper: value(r.recalculated_upper), consistent: r.consistent})
    }
  }
  if (input.missing) {
    const n = overall.n - input.sumN
    missing.push({quantity: "n", digits: null, lower: n, upper: n, feasible: true})
    for (const [quantity, prefix, digits] of [["Mean", "E2: M_miss", input.meanDigits], ["SD", "E2: SD_miss", input.sdDigits]]) {
      const r = fromR("missing", prefix)
      if (r) missing.push({quantity, digits, lower: value(r.recalculated_lower), upper: value(r.recalculated_upper), feasible: r.consistent, single: n === 1})
    }
  }
  return {input, totals, missing, versions, runAt: new Date()}
}

function sgResults(run) {
  const {input, totals, missing, versions} = run
  const fmt = (x, digits) => !Number.isFinite(x) ? "" : (digits === null ? String(x) : x.toFixed(Math.min(digits + 2, 6))).replace("-", "−")
  const range = (lo, hi, digits) => !Number.isFinite(lo) || !Number.isFinite(hi) ? "impossible" : lo === hi ? fmt(lo, digits) : `${fmt(lo, digits)} to ${fmt(hi, digits)}`
  const badge = ([label, kind]) => html`<span class="sc-verdict sc-verdict-${kind}">${label}</span>`
  const totalVerdict = r => r.consistent === null ? ["Not checked", "na"] : r.consistent ? ["Consistent", "ok"] : ["Inconsistent", "bad"]
  const missingVerdict = r => {
    if (r.quantity === "n") return ["Possible", "ok"]
    if (r.quantity === "Mean") return r.feasible === null ? ["Not checked", "na"] : r.feasible ? ["Possible", "ok"] : ["Impossible", "bad"]
    return r.feasible ? ["Possible", "ok"] : ["Impossible", "bad"]
  }
  const missingReason = r => {
    if (r.quantity === "Mean" && r.feasible === null) return "Enter the scale's minimum and maximum to check it."
    if (r.quantity === "Mean" && !r.feasible) return "It is outside the range of the scale."
    if (r.quantity === "Mean") return "It is within the range of the scale."
    if (r.quantity === "SD" && r.single) return r.feasible
      ? "A single participant has no SD, but the reported SD leaves exactly the variability that one participant at the implied mean would add."
      : "A single participant has no SD, and the reported SD does not leave the variability that one participant at the implied mean would add."
    if (r.quantity === "SD" && !Number.isFinite(r.lower)) return "The implied variance is negative for every rounding of the reported values, and no real group can have a negative variance."
    if (r.quantity === "SD" && !r.feasible) return "It is larger than any group of this size and mean could have on this scale."
    if (r.quantity === "SD") return input.useScale ? "It is a real number, and no larger than this group could have on this scale." : "It is a real number. Enter the scale's minimum and maximum to also check that it is not too large for the scale."
    return ""
  }
  const label = q => q === "N" ? html`<em>N</em>` : q === "n" ? html`<em>n</em>` : q === "SD" ? html`<em>SD</em>` : "Mean"
  const blue = "#2780e3", orange = "#d9622b"

  const totalsSection = totals.length === 0 ? "" : html`<div>
    <h5 class="sg-heading">The whole sample, recalculated from the subgroups</h5>
    <div class="table-responsive"><table class="table table-sm sg-table">
      <thead><tr><th></th><th>Reported</th><th>Recalculated from the subgroups</th><th>Verdict</th></tr></thead>
      <tbody>${totals.map(r => html`<tr><td>${label(r.quantity)}</td><td>${r.reported || "–"}</td><td>${range(r.lower, r.upper, r.digits)}</td><td>${badge(totalVerdict(r))}</td></tr>`)}</tbody>
    </table></div>
    ${totals.map(r => sgIntervalPlot(r.quantity, r.digits, [
      ...(Number.isFinite(r.reportedLower) ? [{name: "Reported", lower: r.reportedLower, upper: r.reportedUpper, color: orange}] : []),
      {name: "Recalculated", lower: r.lower, upper: r.upper, color: blue}
    ]))}
    <div class="sc-status">Orange is the reported value, as the range of values that round to it, and blue is the range of values that the subgroups can produce, given the rounding of their own values. Sample sizes are exact, so they are dots. Where orange and blue overlap, the reported value is consistent with the subgroups.</div>
  </div>`

  const sumText = `The subgroups' ns add up to ${input.sumN}`
  const anyInconsistent = totals.some(r => r.consistent === false)
  const missingSection = input.missing ? html`<div>
    <h5 class="sg-heading">An unreported subgroup that would make the reported values consistent</h5>
    <p class="sg-text">${sumText}, which is ${input.overall.n - input.sumN} fewer than the reported <em>N</em> of ${input.overall.n}. If the difference is a single subgroup that was left out of the breakdown, it would need to have these values:</p>
    <div class="table-responsive"><table class="table table-sm sg-table">
      <thead><tr><th></th><th>Implied value</th><th>Verdict</th><th></th></tr></thead>
      <tbody>${missing.map(r => html`<tr><td>${label(r.quantity)}</td><td>${r.single && r.quantity === "SD" ? "not defined" : range(r.lower, r.upper, r.digits)}</td><td>${badge(missingVerdict(r))}</td><td class="sg-reason">${missingReason(r)}</td></tr>`)}</tbody>
    </table></div>
    ${missing.filter(r => r.quantity !== "n" && Number.isFinite(r.lower)).map(r => sgIntervalPlot(r.quantity, r.digits,
      [{name: "Implied", lower: r.lower, upper: r.upper, color: blue}],
      r.quantity === "Mean" && input.useScale ? {lower: input.scale.min, upper: input.scale.max} : null))}
    ${missing.some(r => r.quantity === "Mean" && input.useScale) ? html`<div class="sc-status">The grey band is the range of the scale.</div>` : ""}
    <div class="sc-status">${missing.every(r => missingVerdict(r)[1] !== "bad")
      ? "Such a subgroup is possible, so a missing subgroup could explain the discrepancy. Check whether the article says how many participants were left out of the breakdown, and why."
      : "No real subgroup could have these values, so a single missing subgroup cannot explain the discrepancy."}</div>
  </div>`
    : input.overall.n !== null && input.sumN > input.overall.n ? html`<p class="sg-text">${sumText}, which is ${input.sumN - input.overall.n} more than the reported <em>N</em> of ${input.overall.n}, so no unreported subgroup can explain the difference: the subgroups include more participants than the whole sample.</p>`
    : input.overall.n !== null && anyInconsistent ? html`<p class="sg-text">${sumText}, the same as the reported <em>N</em>, so a missing subgroup cannot explain the inconsistencies.</p>`
    : input.overall.n !== null && totals.length === 0 ? html`<p class="sg-text">${sumText}, the same as the reported <em>N</em>, so there is no unreported subgroup.</p>`
    : ""

  const used = ["Sample sizes were taken as exact counts"]
  if (input.useMeans) used.push(`means as ${input.rounding === "truncate" ? "truncated" : "rounded"} to ${input.meanDigits} decimal place${input.meanDigits === 1 ? "" : "s"}`)
  if (input.useSds) used.push(`SDs to ${input.sdDigits}`)
  const usedText = used.length === 1 ? used[0] : `${used.slice(0, -1).join(", ")}${used.length > 2 ? "," : ""} and ${used.at(-1)}`
  const functions = [input.total && input.useMeans ? "recalc_total_from_subgroups()" : null, input.missingMean ? "recalc_missing_subgroup()" : null].filter(f => f)
  const how = versions
    ? html`Calculated by ${functions.length === 2 ? html`<code>${functions[0]}</code> and <code>${functions[1]}</code>` : html`<code>${functions[0]}</code>`} from recalc ${versions.recalcVersion} (commit <a href="https://github.com/ianhussey/recalc/tree/${recalcCommit}">${recalcCommit.slice(0, 7)}</a>), ${versions.rVersion}, running in webR ${webrVersion}.`
    : "Only sample sizes were entered, so no R was needed: the whole sample's N is the sum of the subgroups' ns."
  return html`<div>
    ${totalsSection}
    ${missingSection}
    ${input.notes.map(n => html`<div class="sc-status sg-note">${n}</div>`)}
    <div class="sc-status">${how} ${usedText}.</div>
    <div class="sb-buttons sc-download">${sgDownloadButton(run)}</div>
  </div>`
}

function sgIntervalPlot(quantity, digits, bars, band = null) {
  // one small plot per quantity, zoomed in on the bars, which are often too narrow to see on the scale's full range.
  // a bar of zero width, such as a sample size, is drawn as a dot
  const ends = [...bars.flatMap(b => [b.lower, b.upper]), ...(band ? [band.lower, band.upper] : [])]
  const [lo, hi] = d3.extent(ends)
  const span = Math.max(hi - lo, digits === null ? 4 : 4 * 10 ** -digits)
  const centre = (lo + hi) / 2
  const domain = [Math.min(lo, centre - span / 2) - 0.1 * span, Math.max(hi, centre + span / 2) + 0.1 * span]
  return html`<div class="sg-plot">
    <div class="sg-plot-title">${quantity === "N" ? html`<em>N</em>` : quantity === "n" ? html`<em>n</em>` : quantity === "SD" ? html`<em>SD</em>` : "Mean"}</div>
    ${Plot.plot({
      width, height: 30 + 32 * bars.length, marginLeft: 110, marginRight: 20, marginBottom: 25,
      x: {domain, label: null, grid: true, tickFormat: digits === null ? "d" : undefined},
      y: {domain: bars.map(b => b.name), label: null, tickSize: 0, padding: 0.3},
      style: {fontSize: "13px"},
      marks: [
        band ? Plot.rectX([band], {x1: "lower", x2: "upper", fill: "currentColor", fillOpacity: 0.08}) : null,
        Plot.link(bars.filter(b => b.upper > b.lower), {x1: "lower", x2: "upper", y1: "name", y2: "name", stroke: "color", strokeWidth: 10}),
        Plot.dot(bars.filter(b => b.upper === b.lower), {x: "lower", y: "name", fill: "color", r: 6})
      ]
    })}
  </div>`
}

function sgDownloadButton(run) {
  // the inputs, the results, and the settings and versions used, one row per subgroup and per result
  const {input, totals, missing, versions, runAt} = run
  const columns = ["section", "row", "subgroup", "n", "mean", "sd", "quantity", "reported", "reported_lower", "reported_upper",
    "recalculated_lower", "recalculated_upper", "verdict", "mean_digits", "sd_digits", "rounding", "scale_min", "scale_max",
    "recalc_version", "recalc_commit", "r_version", "webr_version", "run_at"]
  const settings = {mean_digits: input.useMeans ? input.meanDigits : "", sd_digits: input.useSds ? input.sdDigits : "",
    rounding: input.rounding, scale_min: input.useScale ? input.scale.min : "", scale_max: input.useScale ? input.scale.max : "",
    recalc_version: versions?.recalcVersion ?? "", recalc_commit: versions ? recalcCommit : "", r_version: versions?.rVersion ?? "",
    webr_version: versions ? webrVersion : "", run_at: runAt.toISOString()}
  const finite = x => Number.isFinite(x) ? x : ""
  const records = [
    ...input.rows.map(r => ({section: "subgroup", row: r.row, subgroup: r.label, n: r.n, mean: r.m, sd: r.sd, ...settings})),
    {section: "whole sample, as reported", n: input.overall.text.n, mean: input.overall.text.m, sd: input.overall.text.sd, ...settings},
    ...totals.map(r => ({section: "whole sample, recalculated", quantity: r.quantity, reported: r.reported, reported_lower: finite(r.reportedLower),
      reported_upper: finite(r.reportedUpper), recalculated_lower: r.lower, recalculated_upper: r.upper,
      verdict: r.consistent === null ? "Not checked" : r.consistent ? "Consistent" : "Inconsistent", ...settings})),
    ...missing.map(r => ({section: "implied unreported subgroup", quantity: r.quantity, recalculated_lower: finite(r.lower), recalculated_upper: finite(r.upper),
      verdict: r.quantity === "Mean" && r.feasible === null ? "Not checked" : r.feasible ? "Possible" : "Impossible", ...settings}))
  ]
  return csvDownloadButton(columns, records, "subgroup_results", runAt)
}
